from sqlalchemy import Column, Integer, String, ForeignKey, Date, Time, Boolean, DateTime
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import relationship
from datetime import datetime

Base = declarative_base()

class Customer(Base):
    __tablename__ = 'customers'
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    name = Column(String(21))  # 7個中文字
    address = Column(String(90))  # 30個中文字
    abbreviation = Column(String(30), unique=True)  # 10個中文字
    category = Column(String(10))  # 東洋/診所/其他
    notes = Column(String(30))  # 10個中文字

class Driver(Base):
    __tablename__ = 'drivers'
    
    id = Column(Integer, primary_key=True)  # 6位數字
    name = Column(String(18))  # 6個中文字
    plate_number = Column(String(8))
    brand = Column(String(8))
    model = Column(String(8))

class FixedRoute(Base):
    __tablename__ = 'fixed_routes'
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    weekdays = Column(String(20))  # 星期一~星期六
    time = Column(Time, nullable=False)
    start_point = Column(String(30), ForeignKey('customers.abbreviation'))
    via_point = Column(String(30), ForeignKey('customers.abbreviation'), nullable=True)
    end_point = Column(String(30), ForeignKey('customers.abbreviation'))
    base_fare = Column(Integer)
    surcharge = Column(Integer)
    actual_fare = Column(Integer)
    category = Column(String(10))
    driver_id = Column(Integer, ForeignKey('drivers.id'))

class Schedule(Base):
    __tablename__ = 'schedules'
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    date = Column(Date, nullable=False)
    time = Column(Time, nullable=False)
    start_point = Column(String(30), ForeignKey('customers.abbreviation'))
    via_point = Column(String(30), ForeignKey('customers.abbreviation'), nullable=True)
    end_point = Column(String(30), ForeignKey('customers.abbreviation'))
    base_fare = Column(Integer)
    surcharge = Column(Integer)
    actual_fare = Column(Integer)
    category = Column(String(10))
    driver_id = Column(Integer, ForeignKey('drivers.id'), nullable=True)
    fixed_route_id = Column(Integer, ForeignKey('fixed_routes.id'), nullable=True)
    status = Column(String(10), default='準備')
    is_temporary = Column(Boolean, default=False)

class CompletedSchedule(Base):
    __tablename__ = 'completed_schedules'
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    date = Column(Date, nullable=False)
    time = Column(Time, nullable=False)
    start_point = Column(String(30))
    via_point = Column(String(30), nullable=True)
    end_point = Column(String(30))
    base_fare = Column(Integer)
    surcharge = Column(Integer)
    actual_fare = Column(Integer)
    category = Column(String(10))
    driver_id = Column(Integer)
    driver_name = Column(String(18))
    completion_time = Column(DateTime, default=datetime.now)
    is_temporary = Column(Boolean)
